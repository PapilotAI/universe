\section{Introduction}
Let $f\colon \mathbb{R}\to\mathbb{R}$ be continuous. We show that every such $f$ on $[a,b]$ attains a maximum.

\begin{definition}
A function $f$ is \emph{bounded} on $S$ if there is $M$ with $|f(x)|\le M$ for all $x\in S$.
\end{definition}

\begin{lemma}\label{lem:bounded}
A continuous function on $[a,b]$ is bounded.
\end{lemma}

\begin{theorem}
A continuous function on $[a,b]$ attains its maximum.
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:bounded}, $s=\sup_{[a,b]} f$ is finite; compactness gives a point where $f=s$.
\end{proof}

\begin{remark}
The same argument works on any compact space.
\end{remark}
