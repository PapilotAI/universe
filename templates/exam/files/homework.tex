\question[10] Compute $\displaystyle\int_0^1 x^2\,dx$.
\begin{solution}
  $\int_0^1 x^2\,dx = \left[\tfrac{x^3}{3}\right]_0^1 = \tfrac13$.
\end{solution}

\question Let $A = \begin{pmatrix} 2 & 1 \\ 1 & 2 \end{pmatrix}$.
\begin{parts}
  \part[5] Find the eigenvalues of $A$.
  \begin{solution} $\lambda = 1$ and $\lambda = 3$. \end{solution}
  \part[5] Is $A$ positive definite? Explain.
  \begin{solution} Yes: both eigenvalues are positive. \end{solution}
\end{parts}
